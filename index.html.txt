  <script>
    // 1. DADOS DE DEMONSTRAÇÃO
    const INITIAL_USERS = [
      { email: 'aluno@kognus.com', pass: '123456', name: 'Lucas Aluno', role: 'aluno' },
      { email: 'instrutor@kognus.com', pass: '123456', name: 'Renato Mestre', role: 'instrutor' }
    ];

    const INITIAL_COURSES = [
      { id: 1, title: 'Colorimetria Avançada e Tonalização', category: 'Cabelo', price: 297, instructor: 'Renato Mestre' },
      { id: 2, title: 'Gestão Financeira para Salões', category: 'Negócios', price: 197, instructor: 'Renato Mestre' },
      { id: 3, title: 'Design de Cortes Masculinos Modernos', category: 'Barbearia', price: 147, instructor: 'Camila Hair' }
    ];

    const LESSONS = [
      { id: 1, title: '01. Boas-vindas e Fundamentos', done: true },
      { id: 2, title: '02. Configuração de Materiais', done: false },
      { id: 3, title: '03. Prática em Tempo Real', done: false },
      { id: 4, title: '04. Estratégias de Finalização', done: false }
    ];

    // INICIALIZAÇÃO LOCALSTORAGE
    function initData() {
      if (!localStorage.getItem('kognus_users')) {
        localStorage.setItem('kognus_users', JSON.stringify(INITIAL_USERS));
      }
      if (!localStorage.getItem('kognus_courses')) {
        localStorage.setItem('kognus_courses', JSON.stringify(INITIAL_COURSES));
      }
    }

    // NAVEGAÇÃO ENTRE TELAS
    function navigateTo(screenId) {
      document.querySelectorAll('.screen-view').forEach(el => el.classList.add('hidden'));
      const target = document.getElementById('screen-' + screenId);
      if (target) {
        target.classList.remove('hidden');
        window.scrollTo(0, 0);
      }
      lucide.createIcons();
    }

    // TOAST NOTIFIER
    function showToast(msg) {
      const toast = document.getElementById('toast');
      document.getElementById('toast-message').textContent = msg;
      toast.classList.remove('translate-y-20', 'opacity-0');
      setTimeout(() => {
        toast.classList.add('translate-y-20', 'opacity-0');
      }, 3500);
    }

    // AUTENTICAÇÃO
    function getSession() {
      return JSON.parse(localStorage.getItem('kognus_session'));
    }

    function checkAuth() {
      const session = getSession();
      const navGuest = document.getElementById('nav-guest');
      const navUser = document.getElementById('nav-user');

      if (session) {
        navGuest.classList.add('hidden');
        navUser.classList.remove('hidden');
        document.getElementById('nav-user-name').textContent = `${session.name} (${session.role})`;

        if (session.role === 'instrutor') {
          renderInstructorPanel();
          navigateTo('instrutor-dash');
        } else {
          renderAlunoPanel();
          navigateTo('aluno-dash');
        }
      } else {
        navGuest.classList.remove('hidden');
        navUser.classList.add('hidden');
        navigateTo('landing');
      }
    }

    function handleRegister(e) {
      e.preventDefault();
      const name = document.getElementById('reg-name').value;
      const email = document.getElementById('reg-email').value;
      const pass = document.getElementById('reg-password').value;
      const role = document.querySelector('input[name="reg-role"]:checked').value;

      const users = JSON.parse(localStorage.getItem('kognus_users'));
      if (users.some(u => u.email === email)) {
        showToast('Este e-mail já está cadastrado.');
        return;
      }

      const newUser = { name, email, pass, role };
      users.push(newUser);
      localStorage.setItem('kognus_users', JSON.stringify(users));

      // Login automático pós cadastro
      localStorage.setItem('kognus_session', JSON.stringify(newUser));
      showToast('Conta criada com sucesso! Entrando...');
      checkAuth();
    }

    function handleLogin(e) {
      e.preventDefault();
      const email = document.getElementById('login-email').value;
      const pass = document.getElementById('login-password').value;

      const users = JSON.parse(localStorage.getItem('kognus_users'));
      const user = users.find(u => u.email === email && u.pass === pass);

      if (!user) {
        showToast('E-mail ou senha inválidos.');
        return;
      }

      localStorage.setItem('kognus_session', JSON.stringify(user));
      showToast('Login realizado com sucesso!');
      checkAuth();
    }

    function handleLogout() {
      localStorage.removeItem('kognus_session');
      showToast('Sessão encerrada.');
      checkAuth();
    }

    function openRegisterAs(role) {
      const radio = document.querySelector(`input[name="reg-role"][value="${role}"]`);
      if (radio) radio.checked = true;
      navigateTo('register');
    }

    // RENDERIZADORES
    function renderCoursesLanding() {
      const courses = JSON.parse(localStorage.getItem('kognus_courses'));
      const grid = document.getElementById('landing-courses-grid');
      grid.innerHTML = courses.map(c => `
        <div class="bg-darkCard border border-darkBorder rounded-xl p-5 flex flex-col justify-between hover:border-brand-500/50 transition shadow-lg">
          <div>
            <span class="text-[10px] font-bold px-2 py-0.5 rounded bg-brand-500/10 text-brand-400 uppercase tracking-wider">${c.category}</span>
            <h3 class="font-bold text-lg text-white mt-2">${c.title}</h3>
            <p class="text-xs text-zinc-400 mt-1">Instrutor: ${c.instructor}</p>
          </div>
          <div class="mt-6 flex items-center justify-between pt-4 border-t border-darkBorder">
            <span class="font-extrabold text-xl text-white">R$ ${c.price.toFixed(2)}</span>
            <button onclick="navigateTo('login')" class="bg-zinc-800 hover:bg-zinc-700 text-white text-xs px-3 py-1.5 rounded-lg transition font-medium">
              Acessar
            </button>
          </div>
        </div>
      `).join('');
    }

    function renderAlunoPanel() {
      const list = document.getElementById('lms-lessons-list');
      list.innerHTML = LESSONS.map(lesson => `
        <div class="flex items-center justify-between p-3 rounded-lg bg-zinc-900/60 border border-darkBorder text-xs cursor-pointer hover:border-brand-500 transition" onclick="selectLesson('${lesson.title}')">
          <div class="flex items-center gap-2">
            <input type="checkbox" ${lesson.done ? 'checked' : ''} class="rounded border-zinc-700 text-brand-600 focus:ring-0">
            <span class="${lesson.done ? 'line-through text-zinc-500' : 'text-zinc-200'}">${lesson.title}</span>
          </div>
          <i data-lucide="play-circle" class="w-4 h-4 text-zinc-400"></i>
        </div>
      `).join('');
      lucide.createIcons();
    }

    function selectLesson(title) {
      document.getElementById('lms-lesson-title').textContent = title;
    }

    function renderInstructorPanel() {
      const courses = JSON.parse(localStorage.getItem('kognus_courses'));
      const tbody = document.getElementById('instructor-courses-tbody');
      tbody.innerHTML = courses.map(c => `
        <tr class="hover:bg-zinc-900/30">
          <td class="px-6 py-4 font-medium text-white">${c.title}</td>
          <td class="px-6 py-4">${c.category}</td>
          <td class="px-6 py-4">R$ ${c.price.toFixed(2)}</td>
          <td class="px-6 py-4"><span class="px-2 py-0.5 rounded-full text-[10px] bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">Publicado</span></td>
        </tr>
      `).join('');
    }

    function toggleModalCourse(show) {
      const modal = document.getElementById('modal-course');
      modal.classList.toggle('hidden', !show);
      modal.classList.toggle('flex', show);
    }

    function handleCreateCourse(e) {
      e.preventDefault();
      const session = getSession();
      const title = document.getElementById('course-title').value;
      const category = document.getElementById('course-cat').value;
      const price = parseFloat(document.getElementById('course-price').value);

      const courses = JSON.parse(localStorage.getItem('kognus_courses'));
      courses.push({
        id: Date.now(),
        title,
        category,
        price,
        instructor: session.name
      });

      localStorage.setItem('kognus_courses', JSON.stringify(courses));
      toggleModalCourse(false);
      showToast('Novo curso publicado com sucesso!');
      renderInstructorPanel();
      renderCoursesLanding();
    }
